\newcommand{\resumeItem}[1]{
  \item\small{

\newcommand{\resumeSubheading}[4]{
  \vspace{-2pt}\item

      \textit{\small#3} & \textit{\small #4} \\

% Content macros (the parser knows exactly these)

% \variant{full text}{1-page text}
\newcommand{\variant}[2]{\ifdefined\onepage#2\else#1\fi}
% \fullonly{only in the full PDF}
\newcommand{\fullonly}[1]{\ifdefined\onepage\else#1\fi}

% A separator printed between items of a list, never before the first one.
\newif\iffirstinlist
\newcommand{\listsep}[1]{\iffirstinlist\global\firstinlistfalse\else\unskip#1\fi}

% \resumeSummary{text}: must come before \resumeHeader.
\newcommand{\resumeSummary}[1]{\gdef\summarytext{#1}}

% \resumeContact{label}{url}: only inside \resumeHeader. An empty url prints plain text.
\newcommand{\resumeContact}[2]{%
  \listsep{\space$|$\space}%
  \if\relax\detokenize{#2}\relax #1\else\href{#2}{\underline{#1}}\fi
}

% \resumeHeader{name}{contacts}: the name, the contact line and the summary.
\newcommand{\resumeHeader}[2]{%
  \begin{center}
    \textbf{\Huge \scshape #1} \\ \ifdefined\onepage\vspace{2pt}\else\vspace{1pt}\fi
    \small \global\firstinlisttrue #2
    \ifdefined\onepage \\ \vspace{2pt} \small \summarytext \fi
  \end{center}
  \ifdefined\onepage
    \vspace{-14pt}
  \else
    \section{Professional Summary}
      \summarytext\par
  \fi
}

% \resumeProject{name}{description}, optionally followed by an item list.
\newcommand{\resumeProject}[2]{\resumeProjectHeading{\textbf{#1} $|$ \emph{#2}}{}}

% \resumeAchievements{ \resumeAchievement{title}{detail} ... }
\newcommand{\resumeAchievements}[1]{%
  \ifdefined\onepage
    \begin{itemize}[leftmargin=0.15in, label={}]
      \small{\item{\global\firstinlisttrue #1}}
    \end{itemize}
    \vspace{-12pt}%
  \else
    \resumeSubHeadingListStart #1 \resumeSubHeadingListEnd%
  \fi
}
\newcommand{\resumeAchievement}[2]{%
  \ifdefined\onepage
    \listsep{\space$|$\space}\textbf{#1}{, #2}%
  \else
    \resumeProjectHeading{\textbf{#1} $|$ \emph{#2}}{}%
  \fi
}

% \resumeSkills{ \resumeSkill{label}{items} ... }
\newcommand{\resumeSkills}[1]{%
  \begin{itemize}[leftmargin=0.15in, label={}]
    \small{\item{\global\firstinlisttrue #1}}
  \end{itemize}
}
\newcommand{\resumeSkill}[2]{\listsep{\\}\textbf{#1}{: #2}}
